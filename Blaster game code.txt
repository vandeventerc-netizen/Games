    <script>
        // --- Audio Synthesizer (Web Audio API) ---
        let audioCtx = null;
        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function playSound(type) {
            if (!audioCtx) return;
            try {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                const now = audioCtx.currentTime;

                if (type === 'shoot') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(300, now);
                    osc.frequency.exponentialRampToValueAtTime(600, now + 0.1);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.1);
                    osc.start(now);
                    osc.stop(now + 0.1);
                } else if (type === 'hit') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(600, now);
                    osc.frequency.exponentialRampToValueAtTime(1000, now + 0.15);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.15);
                    osc.start(now);
                    osc.stop(now + 0.15);
                } else if (type === 'miss') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(180, now);
                    osc.frequency.exponentialRampToValueAtTime(80, now + 0.25);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.25);
                    osc.start(now);
                    osc.stop(now + 0.25);
                }
            } catch(e) {
                console.error("Audio error:", e);
            }
        }

        // --- Custom Images Storage ---
        let customImages = [];

        document.getElementById('custom-image-input').addEventListener('change', (e) => {
            const files = e.target.files;
            const previewContainer = document.getElementById('upload-preview');
            
            for (let file of files) {
                const reader = new FileReader();
                reader.onload = (event) => {
                    const img = new Image();
                    img.src = event.target.result;
                    img.onload = () => {
                        customImages.push(img);
                        document.getElementById('custom-count').textContent = customImages.length;
                        
                        const thumb = document.createElement('div');
                        thumb.className = 'w-10 h-10 rounded-lg overflow-hidden border border-blue-200 shadow-sm flex-shrink-0';
                        thumb.innerHTML = `<img src="${event.target.result}" class="w-full h-full object-cover">`;
                        previewContainer.appendChild(thumb);
                    };
                };
                reader.readAsDataURL(file);
            }
        });

        // --- Game Setup ---
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');

        let score = 0;
        let lives = 5;
        let highScore = localStorage.getItem('copilot_shooter_highscore') || 0;
        let gameState = 'MENU'; // MENU, PLAYING, GAMEOVER
        let spawnTimer = 0;
        let spawnInterval = 70;
        let gameSpeedMultiplier = 1;

        document.getElementById('high-score-val').textContent = highScore;

        // Player (Gun)
        const player = {
            x: window.innerWidth / 2,
            y: window.innerHeight - 90,
            width: 80,
            height: 70,
            targetX: window.innerWidth / 2,
            speed: 0.15
        };

        let items = [];
        let projectiles = [];
        let particles = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            player.y = canvas.height - 90;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        window.addEventListener('mousemove', (e) => {
            if (gameState === 'PLAYING') {
                player.targetX = e.clientX;
            }
        });

        window.addEventListener('touchmove', (e) => {
            if (gameState === 'PLAYING' && e.touches.length > 0) {
                player.targetX = e.touches[0].clientX;
            }
        }, { passive: true });

        function shootProjectile() {
            if (gameState !== 'PLAYING') return;
            projectiles.push({
                x: player.x,
                y: player.y - 30,
                radius: 8,
                vy: -12
            });
            playSound('shoot');
        }

        window.addEventListener('click', () => {
            shootProjectile();
        });

        window.addEventListener('keydown', (e) => {
            if (e.code === 'Space') {
                e.preventDefault();
                shootProjectile();
            }
        });

        // Falling Target Class
        class FallingTarget {
            constructor() {
                this.x = Math.random() * (canvas.width - 140) + 70;
                this.y = -60;
                this.vy = (Math.random() * 2 + 2) * gameSpeedMultiplier;
                this.rotation = 0;
                this.rotSpeed = (Math.random() - 0.5) * 0.04;

                this.isEvil = Math.random() < 0.28;

                if (this.isEvil) {
                    this.radius = 32;
                    const evils = ['💣', '😈', '☠️', '⚡'];
                    this.emoji = evils[Math.floor(Math.random() * evils.length)];
                } else if (customImages.length > 0 && Math.random() > 0.2) {
                    this.isCustom = true;
                    this.image = customImages[Math.floor(Math.random() * customImages.length)];
                    this.radius = 36;
                } else {
                    this.isCustom = false;
                    this.radius = 30;
                    const defaults = ['🏆', '⭐', '🍎', '🥇', '💎'];
                    this.emoji = defaults[Math.floor(Math.random() * defaults.length)];
                }
            }

            update() {
                this.y += this.vy;
                this.rotation += this.rotSpeed;
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rotation);
                if (this.isCustom && this.image) {
                    ctx.imageSmoothingEnabled = true;
                    ctx.imageSmoothingQuality = 'high';

                    ctx.save();
                    ctx.beginPath();
                    ctx.arc(0, 0, this.radius, 0, Math.PI * 2);
                    ctx.closePath();
                    ctx.fillStyle = '#ffffff';
                    ctx.fill();
                    ctx.clip();
                    ctx.drawImage(this.image, -this.radius, -this.radius, this.radius * 2, this.radius * 2);
                    ctx.restore();

                    ctx.strokeStyle = '#2563eb';
                    ctx.lineWidth = 3;
                    ctx.beginPath();
                    ctx.arc(0, 0, this.radius, 0, Math.PI * 2);
                    ctx.stroke();
                } else {
                    ctx.font = '40px sans-serif';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText(this.emoji, 0, 0);
                }
                ctx.restore();
            }
        }

        // Particle Class for Hit Effects
        class Particle {
            constructor(x, y, text, color) {
                this.x = x;
                this.y = y;
                this.text = text;
                this.color = color;
                this.alpha = 1;
                this.vy = -2.5;
                this.scale = 1;
            }
            update() {
                this.y += this.vy;
                this.alpha -= 0.03;
                this.scale += 0.02;
            }
            draw() {
                ctx.save();
                ctx.globalAlpha = Math.max(0, this.alpha);
                ctx.font = 'bold 20px Poppins, sans-serif';
                ctx.fillStyle = this.color;
                ctx.textAlign = 'center';
                ctx.fillText(this.text, this.x, this.y);
                ctx.restore();
            }
        }

        function startGame() {
            initAudio();
            score = 0;
            lives = 5;
            gameSpeedMultiplier = 1;
            spawnInterval = 70;
            items = [];
            projectiles = [];
            particles = [];
            player.x = canvas.width / 2;
            player.targetX = canvas.width / 2;

            document.getElementById('menu-screen').classList.add('hidden');
            document.getElementById('gameover-screen').classList.add('hidden');
            document.getElementById('hud').classList.remove('hidden');
            
            updateHUD();
            gameState = 'PLAYING';
        }

        function gameOver() {
            gameState = 'GAMEOVER';
            document.getElementById('hud').classList.add('hidden');
            document.getElementById('gameover-screen').classList.remove('hidden');
            document.getElementById('final-score').textContent = score;

            if (score > highScore) {
                highScore = score;
                localStorage.setItem('copilot_shooter_highscore', highScore);
                document.getElementById('high-score-val').textContent = highScore;
            }
            document.getElementById('best-score').textContent = highScore;
        }

        function updateHUD() {
            document.getElementById('score-val').textContent = score;
            let hearts = '';
            for (let i = 0; i < 5; i++) {
                hearts += i < lives ? '❤️ ' : '🖤 ';
            }
            document.getElementById('lives-container').textContent = hearts.trim();
        }

        function loop() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            ctx.imageSmoothingEnabled = true;
            ctx.imageSmoothingQuality = 'high';

            if (gameState === 'PLAYING') {
                player.x += (player.targetX - player.x) * player.speed;
                const halfWidth = player.width / 2;
                if (player.x < halfWidth) player.x = halfWidth;
                if (player.x > canvas.width - halfWidth) player.x = canvas.width - halfWidth;

                spawnTimer++;
                if (spawnTimer >= spawnInterval) {
                    spawnTimer = 0;
                    items.push(new FallingTarget());
                    if (spawnInterval > 30) spawnInterval -= 0.6;
                    gameSpeedMultiplier += 0.006;
                }

                // Update Projectiles
                for (let i = projectiles.length - 1; i >= 0; i--) {
                    let proj = projectiles[i];
                    proj.y += proj.vy;

                    ctx.save();
                    ctx.font = '24px sans-serif';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText('✨', proj.x, proj.y);
                    ctx.restore();

                    if (proj.y < -20) {
                        projectiles.splice(i, 1);
                    }
                }

                // Update Falling Targets
                for (let i = items.length - 1; i >= 0; i--) {
                    let item = items[i];
                    item.update();
                    item.draw();

                    let hit = false;
                    for (let j = projectiles.length - 1; j >= 0; j--) {
                        let p = projectiles[j];
                        let dist = Math.hypot(p.x - item.x, p.y - item.y);
                        if (dist < item.radius + p.radius) {
                            hit = true;
                            projectiles.splice(j, 1);
                            
                            if (item.isEvil) {
                                lives--;
                                playSound('miss');
                                particles.push(new Particle(item.x, item.y, '-1 💣', '#e11d48'));
                                updateHUD();
                                if (lives <= 0) {
                                    gameOver();
                                }
                            } else {
                                score += 1;
                                playSound('hit');
                                particles.push(new Particle(item.x, item.y, '+1 🏆', '#2563eb'));
                                updateHUD();
                            }
                            break;
                        }
                    }

                    if (hit) {
                        items.splice(i, 1);
                        continue;
                    }

                    // Check if item hit bottom
                    if (item.y > canvas.height + 40) {
                        if (item.isEvil) {
                            // Evil object reached bottom/player base
                            lives--;
                            playSound('miss');
                            particles.push(new Particle(item.x, canvas.height - 50, '-1 💥', '#e11d48'));
                            updateHUD();
                            if (lives <= 0) {
                                gameOver();
                            }
                        } else {
                            // Good item missed
                            lives--;
                            playSound('miss');
                            particles.push(new Particle(item.x, canvas.height - 50, '-1 💔', '#e11d48'));
                            updateHUD();
                            if (lives <= 0) {
                                gameOver();
                            }
                        }
                        items.splice(i, 1);
                    }
                }

                // Update Particles
                for (let i = particles.length - 1; i >= 0; i--) {
                    let part = particles[i];
                    part.update();
                    part.draw();
                    if (part.alpha <= 0) {
                        particles.splice(i, 1);
                    }
                }

                drawPlayer();
            }

            requestAnimationFrame(loop);
        }

        function drawPlayer() {
            ctx.save();
            ctx.translate(player.x, player.y);

            // HondaJet Custom Canvas Drawing (Pointing Upwards)
            ctx.imageSmoothingEnabled = true;

            // Wings
            ctx.fillStyle = '#e2e8f0';
            ctx.beginPath();
            ctx.moveTo(0, -10);
            ctx.lineTo(-52, 15);
            ctx.lineTo(-46, 26);
            ctx.lineTo(0, 15);
            ctx.lineTo(46, 26);
            ctx.lineTo(52, 15);
            ctx.closePath();
            ctx.fill();
            ctx.strokeStyle = '#cbd5e1';
            ctx.lineWidth = 2;
            ctx.stroke();

            // Fuselage (Sleek aerodynamic white body)
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.ellipse(0, 0, 14, 46, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.strokeStyle = '#94a3b8';
            ctx.lineWidth = 2;
            ctx.stroke();

            // HondaJet Signature Blue Accent Stripe
            ctx.fillStyle = '#2563eb';
            ctx.beginPath();
            ctx.ellipse(0, 5, 12, 26, 0, 0, Math.PI * 2);
            ctx.fill();

            // Cockpit Window (Dark tinted)
            ctx.fillStyle = '#0f172a';
            ctx.beginPath();
            ctx.ellipse(0, -23, 8, 12, 0, 0, Math.PI * 2);
            ctx.fill();

            // Distinctive Over-The-Wing Engines (OTWEM) mounted on pylons
            // Left Engine
            ctx.fillStyle = '#64748b';
            ctx.beginPath();
            ctx.roundRect(-33, -3, 13, 25, 6);
            ctx.fill();
            ctx.fillStyle = '#1e293b';
            ctx.beginPath();
            ctx.ellipse(-26, -3, 6.5, 4, 0, 0, Math.PI * 2);
            ctx.fill();

            // Right Engine
            ctx.fillStyle = '#64748b';
            ctx.beginPath();
            ctx.roundRect(20, -3, 13, 25, 6);
            ctx.fill();
            ctx.fillStyle = '#1e293b';
            ctx.beginPath();
            ctx.ellipse(26, -3, 6.5, 4, 0, 0, Math.PI * 2);
            ctx.fill();

            // Tail Fin
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.moveTo(0, 32);
            ctx.lineTo(-16, 48);
            ctx.lineTo(0, 42);
            ctx.lineTo(16, 48);
            ctx.closePath();
            ctx.fill();
            ctx.strokeStyle = '#cbd5e1';
            ctx.lineWidth = 1;
            ctx.stroke();

            // Jet Label Badge
            ctx.fillStyle = 'rgba(15, 23, 42, 0.85)';
            ctx.beginPath();
            ctx.roundRect(-42, 52, 84, 20, 8);
            ctx.fill();

            ctx.fillStyle = '#38bdf8';
            ctx.font = 'bold 10px Poppins, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('HONDAJET', 0, 62);

            ctx.restore();
        }

        document.getElementById('start-btn').addEventListener('click', startGame);
        document.getElementById('restart-btn').addEventListener('click', startGame);

        requestAnimationFrame(loop);
    </script>
